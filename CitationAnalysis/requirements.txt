# BibVik Citation Analysis — Python Dependencies
#
# Install core dependencies:
#   pip install -r requirements.txt
#
# Install with OCR support for scanned PDFs:
#   pip install -e ".[ocr]"
#   (also requires Tesseract at the system level — see README)
#
# Requires Python >= 3.10

# HTTP client for GROBID and Ollama API calls
requests>=2.31.0

# XML parsing for GROBID's TEI-XML output
lxml>=5.1.0

# YAML config file parsing
pyyaml>=6.0.1

# Unicode text handling — transliteration for citekey generation
# (e.g., "Müller" → "muller", "Sindbæk" → "sindbaek")
unidecode>=1.3.8

# Progress bars for long-running batch operations
tqdm>=4.66.0

# Language detection for paper body text
# Detects English, Norwegian, Danish, Swedish, German, French.
# Used for audit tool language strata and downstream context analysis.
# Chose over langdetect for: deterministic results, better accuracy on
# closely related Scandinavian languages, better short-text performance.
lingua-language-detector>=2.0.0

# ---------------------------------------------------------------------------
# Optional: OCR support
# ---------------------------------------------------------------------------
# Needed only if the corpus contains scanned PDFs with no text layer.
# The pipeline detects these automatically (GROBID returns [NO_BLOCKS])
# and applies OCR as a fallback; without this package, those PDFs are
# skipped with a clear error message rather than crashing.
#
# Also requires Tesseract at the system level:
#   macOS:  brew install tesseract
#   Ubuntu: apt install tesseract-ocr
#
# Uncomment to install:
# ocrmypdf>=16.0.0

# Cyrillic → Latin transliteration using ALA-LC tables (American Library
# Association / Library of Congress standard). Used for cross-script
# duplicate detection in _find_duplicate(). Falls back to a hand-rolled
# table if not installed — install for best accuracy.
domovyk>=0.1.0
